{{/* 项目级图片渲染钩子（覆盖 HugoBlox kit 的同名钩子）。 为什么需要它： Goldmark 只把「Markdown 图片语法」交给这个钩子，钩子里如果解析不到资源， kit 会原样输出 。 此时 `![x](assets/x.png)` 里的 `assets/x.png` 是「相对于**页面 URL**」解析的， 而页面 URL 是目录形（/zh-hans/help/epointer-paginator/）， 浏览器会去找 /zh-hans/help/epointer-paginator/assets/x.png → 404。 本钩子在「解析不到资源」时，把相对路径按**源 md 文件所在目录**重写为站点绝对路径， 也就是和 Markdown 编辑器（以及读者的直觉）一致的语义： content//help/epointer-tips.md 中的 assets/a.png → //help/assets/a.png 绝对路径（/ 开头）与远程地址保持 kit 原行为不变。 */}} {{ $destination := .Destination }} {{ $is_remote := strings.HasPrefix $destination "http" }} {{ $is_absolute := strings.HasPrefix $destination "/" }} {{ $caption := .Title | default "" }} {{ $zoom := "true" }} {{ $id := anchorize ($caption | plainify) }} {{ $alt := .Text | default ($caption | plainify) }} {{ $img_class := "" }} {{ $fig_class := "" }} {{ $max_width := "" }} {{ $width := "" }} {{ $height := "" }} {{ $numbered := false }} {{/* Workaround Hugo v0.81 error on Windows when `resources.Get (path.Join "media" )` */}} {{- $img := "" -}} {{- $src := $destination -}} {{- if not $is_remote -}} {{- $img = (.Page.Resources.ByType "image").GetMatch $destination -}} {{- if not $img -}} {{- $img = resources.Get (path.Join "media" $destination) -}} {{- end -}} {{- if and (not $img) (not $is_absolute) -}} {{/* 相对路径按源文件目录重写（见文件头注释） */}} {{- $dir := "" -}} {{- with .Page.File }}{{ $dir = .Dir }}{{ end -}} {{- $dir = replace $dir "\\" "/" -}} {{- $src = path.Join .Page.Site.Home.RelPermalink $dir $destination -}} {{- if not (os.FileExists (path.Join "content" site.Language.Lang $dir $destination)) -}} {{- warnf "render-image: 相对图片路径解析后文件不存在：%q → %q（%s）" $destination $src .Position -}} {{- end -}} {{- end -}} {{- if and (not $img) $is_absolute -}} {{- $src = $destination | relURL -}} {{- end -}} {{- end -}}
